cmake_minimum_required(VERSION 3.25)
project(EndfieldFirstPerson VERSION 0.1.7 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(FirstPerson SHARED src/plugin.cpp)
target_include_directories(FirstPerson PRIVATE src sdk/include)
set_target_properties(FirstPerson PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/first-person")
add_custom_target(package ALL DEPENDS FirstPerson)
foreach(f mod.ini config.ini icon.png controller.lua native.lua zml-package.json)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_CURRENT_SOURCE_DIR}/mod/${f}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/first-person/${f}")
endforeach()
enable_testing()
add_executable(ContractTests tests/contract.cpp)
target_include_directories(ContractTests PRIVATE src sdk/include)
add_test(NAME contract COMMAND ContractTests "$<TARGET_FILE:FirstPerson>")
set(ZML_LUPA_PATH "" CACHE PATH "Path to Lupa installation")
set(ZML_CLIENT_LUA "" CACHE PATH "Path to game Lua scripts")
set(ZML_KEYBINDS_LUA "" CACHE FILEPATH "Path to keybinds API fixture")
set(ZML_SERVICES_EXE "" CACHE FILEPATH "Path to loader services executable")
set(ZML_INSTALL_SCRIPT "" CACHE FILEPATH "Path to installer script")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME package_manifest COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/package.py" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/first-person")
if(ZML_LUPA_PATH)
  add_test(NAME behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/behavior.py" --lupa-dir "${ZML_LUPA_PATH}" --keybinds "${ZML_KEYBINDS_LUA}")
  add_test(NAME native_adapter COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/native.py" --lupa-dir "${ZML_LUPA_PATH}")
endif()
if(ZML_INSTALL_SCRIPT)
  add_test(NAME install_rollback COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/install.py" "${ZML_INSTALL_SCRIPT}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/first-person")
endif()
if(ZML_CLIENT_LUA)
  add_test(NAME client_contract COMMAND ContractTests "$<TARGET_FILE:FirstPerson>" "${ZML_CLIENT_LUA}" "${CMAKE_BINARY_DIR}/client-patched")
  if(ZML_LUPA_PATH)
    add_test(NAME client_syntax COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/syntax.py" "${ZML_LUPA_PATH}" "${CMAKE_BINARY_DIR}/client-patched")
    set_tests_properties(client_syntax PROPERTIES DEPENDS client_contract)
  endif()
endif()
if(ZML_SERVICES_EXE)
  add_test(NAME public_services COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/services.py" "${ZML_SERVICES_EXE}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/first-person")
endif()
